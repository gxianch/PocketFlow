# 高级特性：重试与容错

## 概述

PocketFlow 的 `Node` 类内置了重试机制，帮助处理临时性故障（如网络超时、API 限流等）。同时提供了 `exec_fallback` 方法用于优雅降级。

## 重试机制

### 核心参数

```python
class Node(BaseNode):
    def __init__(self, max_retries=1, wait=0):
        super().__init__()
        self.max_retries = max_retries  # 最大重试次数
        self.wait = wait                # 重试间隔（秒）
```

| 参数 | 类型 | 默认值 | 说明 |
|------|------|--------|------|
| `max_retries` | int | 1 | 最大执行次数（包括首次） |
| `wait` | float | 0 | 重试前等待的秒数 |
| `cur_retry` | int | - | 当前重试次数（运行时，从 0 开始） |

### 源码解析

```python
def _exec(self, prep_res):
    for self.cur_retry in range(self.max_retries):
        try:
            return self.exec(prep_res)
        except Exception as e:
            if self.cur_retry == self.max_retries - 1:
                # 最后一次重试失败，调用降级处理
                return self.exec_fallback(prep_res, e)
            if self.wait > 0:
                time.sleep(self.wait)
```

## 基本用法

### 简单重试

```python
class LLMNode(Node):
    def __init__(self):
        # 最多重试 3 次，每次间隔 1 秒
        super().__init__(max_retries=3, wait=1)

    def exec(self, prompt):
        # 可能因网络问题失败
        response = call_llm_api(prompt)
        return response
```

### 获取当前重试次数

```python
class RetryAwareNode(Node):
    def __init__(self):
        super().__init__(max_retries=5, wait=2)

    def exec(self, data):
        if self.cur_retry > 0:
            print(f"重试第 {self.cur_retry} 次...")
        return process(data)
```

## exec_fallback 降级处理

### 默认行为

```python
def exec_fallback(self, prep_res, exc):
    raise exc  # 默认：重新抛出异常
```

### 自定义降级

```python
class GracefulNode(Node):
    def __init__(self):
        super().__init__(max_retries=3, wait=1)

    def exec(self, data):
        return api_call(data)

    def exec_fallback(self, prep_res, exc):
        # 返回默认值而非抛出异常
        return {
            "success": False,
            "error": str(exc),
            "data": None
        }
```

### 降级策略示例

#### 策略1：返回缓存数据

```python
class CacheFallbackNode(Node):
    cache = {}

    def exec(self, key):
        result = fetch_from_api(key)
        self.cache[key] = result  # 缓存成功结果
        return result

    def exec_fallback(self, key, exc):
        # 尝试返回缓存
        if key in self.cache:
            return {"data": self.cache[key], "from_cache": True}
        raise exc  # 无缓存则抛出异常
```

#### 策略2：使用备用服务

```python
class FallbackServiceNode(Node):
    def exec(self, data):
        return primary_service(data)

    def exec_fallback(self, data, exc):
        print(f"主服务失败: {exc}，切换到备用服务")
        return backup_service(data)
```

#### 策略3：返回默认响应

```python
class DefaultResponseNode(Node):
    DEFAULT_RESPONSE = "抱歉，我暂时无法回答这个问题。"

    def exec(self, prompt):
        return llm_call(prompt)

    def exec_fallback(self, prompt, exc):
        return self.DEFAULT_RESPONSE
```

#### 策略4：记录并继续

```python
class LogAndContinueNode(Node):
    def exec(self, data):
        return process(data)

    def exec_fallback(self, data, exc):
        # 记录错误日志
        logging.error(f"处理失败: {data}, 错误: {exc}")

        # 标记为失败但不中断流程
        return {"status": "failed", "error": str(exc), "input": data}
```

## 异步重试

`AsyncNode` 同样支持重试：

```python
class AsyncNode(Node):
    async def _exec(self, prep_res):
        for self.cur_retry in range(self.max_retries):
            try:
                return await self.exec_async(prep_res)
            except Exception as e:
                if self.cur_retry == self.max_retries - 1:
                    return await self.exec_fallback_async(prep_res, e)
                if self.wait > 0:
                    await asyncio.sleep(self.wait)  # 异步等待
```

### 异步降级示例

```python
class AsyncLLMNode(AsyncNode):
    def __init__(self):
        super().__init__(max_retries=3, wait=2)

    async def exec_async(self, prompt):
        return await async_llm_call(prompt)

    async def exec_fallback_async(self, prompt, exc):
        # 异步降级处理
        return await fallback_llm_call(prompt)
```

## 批处理中的错误处理

### BatchNode 的重试行为

`BatchNode` 对每个 item 独立重试：

```python
class BatchNode(Node):
    def _exec(self, items):
        return [super(BatchNode, self)._exec(i) for i in (items or [])]
```

每个 item 的 `exec` 都有独立的重试次数。

### 批处理降级示例

```python
class RobustBatchNode(BatchNode):
    def __init__(self):
        super().__init__(max_retries=3, wait=1)

    def exec(self, item):
        return process_item(item)

    def exec_fallback(self, item, exc):
        # 单个 item 失败不影响其他
        return {"error": str(exc), "item": item}

    def post(self, shared, prep_res, exec_res):
        # 分离成功和失败
        shared["successes"] = [r for r in exec_res if "error" not in r]
        shared["failures"] = [r for r in exec_res if "error" in r]

        if shared["failures"]:
            print(f"警告: {len(shared['failures'])} 个项目处理失败")
```

## 高级重试策略

### 指数退避

```python
class ExponentialBackoffNode(Node):
    def __init__(self, max_retries=5, initial_wait=1):
        super().__init__(max_retries=max_retries, wait=0)
        self.initial_wait = initial_wait

    def _exec(self, prep_res):
        for self.cur_retry in range(self.max_retries):
            try:
                return self.exec(prep_res)
            except Exception as e:
                if self.cur_retry == self.max_retries - 1:
                    return self.exec_fallback(prep_res, e)

                # 指数退避：1s, 2s, 4s, 8s, ...
                wait_time = self.initial_wait * (2 ** self.cur_retry)
                print(f"等待 {wait_time} 秒后重试...")
                time.sleep(wait_time)
```

### 带抖动的退避

```python
import random

class JitteredBackoffNode(Node):
    def _exec(self, prep_res):
        for self.cur_retry in range(self.max_retries):
            try:
                return self.exec(prep_res)
            except Exception as e:
                if self.cur_retry == self.max_retries - 1:
                    return self.exec_fallback(prep_res, e)

                # 添加随机抖动避免雷群效应
                base_wait = 2 ** self.cur_retry
                jitter = random.uniform(0, base_wait * 0.1)
                time.sleep(base_wait + jitter)
```

### 条件重试

```python
class ConditionalRetryNode(Node):
    def __init__(self):
        super().__init__(max_retries=5, wait=1)

    def _exec(self, prep_res):
        for self.cur_retry in range(self.max_retries):
            try:
                return self.exec(prep_res)
            except RateLimitError as e:
                # 限流错误：等待更长时间
                if self.cur_retry < self.max_retries - 1:
                    time.sleep(60)  # 等待 1 分钟
            except NetworkError as e:
                # 网络错误：正常重试
                if self.cur_retry < self.max_retries - 1:
                    time.sleep(self.wait)
            except ValueError as e:
                # 数据错误：不重试，直接降级
                return self.exec_fallback(prep_res, e)
            except Exception as e:
                if self.cur_retry == self.max_retries - 1:
                    return self.exec_fallback(prep_res, e)
```

## 在 Flow 中的错误处理

### 节点级错误处理

```python
class ErrorHandlingFlow(Flow):
    def _orch(self, shared, params=None):
        try:
            return super()._orch(shared, params)
        except Exception as e:
            shared["flow_error"] = str(e)
            return "error"
```

### 错误恢复节点

```python
class DecideNode(Node):
    def post(self, shared, prep_res, exec_res):
        if exec_res.get("error"):
            return "handle_error"
        return "continue"

class ErrorHandlerNode(Node):
    def exec(self, error_info):
        # 处理错误并决定是否重试整个流程
        if is_retryable(error_info):
            return {"action": "retry"}
        return {"action": "abort", "message": "不可恢复的错误"}

# 流程定义
decide - "handle_error" >> error_handler
decide - "continue" >> next_node
error_handler - "retry" >> decide  # 回到决策点重试
```

## 最佳实践

### 1. 合理设置重试次数

```python
# API 调用：3-5 次通常足够
api_node = APINode(max_retries=3, wait=2)

# 不稳定网络：可以更多
network_node = NetworkNode(max_retries=10, wait=5)

# 数据库操作：较少重试
db_node = DBNode(max_retries=2, wait=1)
```

### 2. 区分可重试和不可重试错误

```python
class SmartRetryNode(Node):
    RETRYABLE_ERRORS = (TimeoutError, ConnectionError, RateLimitError)

    def _exec(self, prep_res):
        for self.cur_retry in range(self.max_retries):
            try:
                return self.exec(prep_res)
            except self.RETRYABLE_ERRORS as e:
                if self.cur_retry == self.max_retries - 1:
                    return self.exec_fallback(prep_res, e)
                time.sleep(self.wait)
            except Exception as e:
                # 其他错误直接降级
                return self.exec_fallback(prep_res, e)
```

### 3. 记录重试日志

```python
class LoggedNode(Node):
    def _exec(self, prep_res):
        for self.cur_retry in range(self.max_retries):
            try:
                result = self.exec(prep_res)
                if self.cur_retry > 0:
                    logging.info(f"第 {self.cur_retry + 1} 次重试成功")
                return result
            except Exception as e:
                logging.warning(f"第 {self.cur_retry + 1} 次执行失败: {e}")
                if self.cur_retry == self.max_retries - 1:
                    logging.error(f"重试耗尽，执行降级处理")
                    return self.exec_fallback(prep_res, e)
                time.sleep(self.wait)
```

### 4. 避免无限循环

```python
class SafeLoopNode(Node):
    MAX_ITERATIONS = 100

    def post(self, shared, prep_res, exec_res):
        iterations = shared.get("iterations", 0) + 1
        shared["iterations"] = iterations

        if iterations >= self.MAX_ITERATIONS:
            logging.error("达到最大迭代次数，强制退出")
            return "force_exit"

        if exec_res.get("success"):
            return "done"
        return "retry"
```
