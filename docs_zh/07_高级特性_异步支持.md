# 高级特性：异步支持

## 概述

PocketFlow 提供完整的异步支持，使用 Python 的 `asyncio` 实现。异步节点特别适合 I/O 密集型任务，如 API 调用、数据库查询等。

## 异步类层次结构

```
AsyncNode                    ─── 异步节点
├── AsyncBatchNode           ─── 异步顺序批处理
└── AsyncParallelBatchNode   ─── 异步并行批处理

AsyncFlow                    ─── 异步流程
├── AsyncBatchFlow           ─── 异步顺序批量流程
└── AsyncParallelBatchFlow   ─── 异步并行批量流程
```

## AsyncNode

### 源码解析

```python
class AsyncNode(Node):
    async def prep_async(self, shared):
        """异步准备阶段"""
        pass

    async def exec_async(self, prep_res):
        """异步执行阶段"""
        pass

    async def exec_fallback_async(self, prep_res, exc):
        """异步降级处理"""
        raise exc

    async def post_async(self, shared, prep_res, exec_res):
        """异步后处理阶段"""
        pass

    async def _exec(self, prep_res):
        for self.cur_retry in range(self.max_retries):
            try:
                return await self.exec_async(prep_res)
            except Exception as e:
                if self.cur_retry == self.max_retries - 1:
                    return await self.exec_fallback_async(prep_res, e)
                if self.wait > 0:
                    await asyncio.sleep(self.wait)

    async def run_async(self, shared):
        """异步运行入口"""
        if self.successors:
            warnings.warn("Node won't run successors. Use AsyncFlow.")
        return await self._run_async(shared)

    async def _run_async(self, shared):
        p = await self.prep_async(shared)
        e = await self._exec(p)
        return await self.post_async(shared, p, e)

    def _run(self, shared):
        raise RuntimeError("Use run_async.")
```

### 基本用法

```python
from pocketflow import AsyncNode
import asyncio

class AsyncLLMNode(AsyncNode):
    async def prep_async(self, shared):
        return shared["prompt"]

    async def exec_async(self, prompt):
        # 异步调用 LLM
        response = await async_llm_call(prompt)
        return response

    async def post_async(self, shared, prep_res, exec_res):
        shared["response"] = exec_res

# 运行
async def main():
    node = AsyncLLMNode()
    shared = {"prompt": "What is AI?"}
    await node.run_async(shared)
    print(shared["response"])

asyncio.run(main())
```

## AsyncFlow

### 源码解析

```python
class AsyncFlow(Flow, AsyncNode):
    async def _orch_async(self, shared, params=None):
        curr = copy.copy(self.start_node)
        p = params or {**self.params}
        last_action = None

        while curr:
            curr.set_params(p)
            # 智能判断：异步节点用 _run_async，同步节点用 _run
            if isinstance(curr, AsyncNode):
                last_action = await curr._run_async(shared)
            else:
                last_action = curr._run(shared)
            curr = copy.copy(self.get_next_node(curr, last_action))

        return last_action

    async def _run_async(self, shared):
        p = await self.prep_async(shared)
        o = await self._orch_async(shared)
        return await self.post_async(shared, p, o)

    async def post_async(self, shared, prep_res, exec_res):
        return exec_res
```

### 混合节点支持

`AsyncFlow` 可以同时包含同步和异步节点：

```python
class SyncNode(Node):
    def exec(self, data):
        return process(data)  # 同步操作

class AsyncAPINode(AsyncNode):
    async def exec_async(self, data):
        return await api_call(data)  # 异步操作

# 混合使用
sync_node = SyncNode()
async_node = AsyncAPINode()

sync_node >> async_node  # AsyncFlow 会正确处理

flow = AsyncFlow(start=sync_node)
await flow.run_async(shared)
```

## 并行批处理

### AsyncBatchNode（顺序）

```python
class AsyncBatchNode(AsyncNode, BatchNode):
    async def _exec(self, items):
        # 顺序执行：一个接一个
        return [await super()._exec(i) for i in items]
```

### AsyncParallelBatchNode（并行）

```python
class AsyncParallelBatchNode(AsyncNode, BatchNode):
    async def _exec(self, items):
        # 并行执行：同时处理所有项
        return await asyncio.gather(
            *(super()._exec(i) for i in items)
        )
```

### 性能对比

```python
import time

# 模拟 1 秒的 API 调用
async def slow_api(item):
    await asyncio.sleep(1)
    return f"processed: {item}"

# 顺序批处理：10 个项目 = 10 秒
class SequentialNode(AsyncBatchNode):
    async def exec_async(self, item):
        return await slow_api(item)

# 并行批处理：10 个项目 = 1 秒
class ParallelNode(AsyncParallelBatchNode):
    async def exec_async(self, item):
        return await slow_api(item)
```

```
顺序执行 (AsyncBatchNode):
┌──────┐┌──────┐┌──────┐┌──────┐
```

### 并行批处理示例

```python
from pocketflow import AsyncParallelBatchNode

class ParallelTranslateNode(AsyncParallelBatchNode):
    async def prep_async(self, shared):
        return shared["texts"]

    async def exec_async(self, text):
        # 每个文本并行翻译
        return await async_translate(text)

# 使用
```

## AsyncBatchFlow 和 AsyncParallelBatchFlow

### AsyncBatchFlow（顺序）

```python
class AsyncBatchFlow(AsyncFlow, BatchFlow):
    async def _run_async(self, shared):
        pr = await self.prep_async(shared) or []
        for bp in pr:
            await self._orch_async(shared, {**self.params, **bp})
        return await self.post_async(shared, pr, None)
```

每次流程按顺序执行。

### AsyncParallelBatchFlow（并行）

```python
class AsyncParallelBatchFlow(AsyncFlow, BatchFlow):
    async def _run_async(self, shared):
        pr = await self.prep_async(shared) or []
        await asyncio.gather(
            *(self._orch_async(shared, {**self.params, **bp}) for bp in pr)
        )
        return await self.post_async(shared, pr, None)
```

所有流程并行执行。

### 并行流程示例

```python
class ParallelProcessFlow(AsyncParallelBatchFlow):
    async def prep_async(self, shared):
        return [{"file": f} for f in shared["files"]]

class AsyncReadNode(AsyncNode):
    async def prep_async(self, shared):
        return self.params["file"]

    async def exec_async(self, filepath):
        async with aiofiles.open(filepath) as f:
            return await f.read()

    async def post_async(self, shared, prep_res, exec_res):
        # 注意：并行写入需要考虑线程安全
        async with shared["lock"]:
            if "contents" not in shared:
                shared["contents"] = {}
            shared["contents"][prep_res] = exec_res

# 使用
async def main():
    read_node = AsyncReadNode()
    flow = ParallelProcessFlow(start=read_node)

    shared = {
        "files": ["a.txt", "b.txt", "c.txt"],
        "lock": asyncio.Lock()  # 保护并发写入
    }
    await flow.run_async(shared)

asyncio.run(main())
```

## 并发控制

### 使用 Semaphore 限制并发

```python
class RateLimitedNode(AsyncParallelBatchNode):
    def __init__(self, max_concurrent=5):
        super().__init__()
        self.semaphore = asyncio.Semaphore(max_concurrent)

    async def exec_async(self, item):
        async with self.semaphore:
            # 最多 5 个并发
            return await api_call(item)
```

### 使用 asyncio.wait 超时控制

```python
class TimeoutNode(AsyncNode):
    async def exec_async(self, data):
        try:
            return await asyncio.wait_for(
                slow_operation(data),
                timeout=30.0  # 30 秒超时
            )
        except asyncio.TimeoutError:
            return {"error": "timeout"}
```

## 最佳实践

### 1. 选择合适的并行度

```python
# ❌ 无限制并发可能导致问题
class BadNode(AsyncParallelBatchNode):
    async def exec_async(self, item):
        return await api_call(item)  # 可能触发 API 限流

# ✅ 限制并发数
class GoodNode(AsyncParallelBatchNode):
    semaphore = asyncio.Semaphore(10)  # 最多 10 个并发

    async def exec_async(self, item):
        async with self.semaphore:
            return await api_call(item)
```

### 2. 并行写入的线程安全

```python
class SafeParallelNode(AsyncParallelBatchNode):
    async def post_async(self, shared, prep_res, exec_res):
        # 使用锁保护共享数据
        async with shared.get("lock", asyncio.Lock()):
            if "results" not in shared:
                shared["results"] = []
            shared["results"].extend(exec_res)
```

### 3. 错误处理

```python
class RobustParallelNode(AsyncParallelBatchNode):
    async def exec_async(self, item):
        try:
            return await process(item)
        except Exception as e:
            # 不让单个错误影响整体
            return {"error": str(e), "item": item}

    async def post_async(self, shared, prep_res, exec_res):
        shared["successes"] = [r for r in exec_res if "error" not in r]
        shared["failures"] = [r for r in exec_res if "error" in r]
```

### 4. 资源清理

```python
class ResourceNode(AsyncNode):
    async def prep_async(self, shared):
        # 创建连接
        shared["connection"] = await create_connection()
        return shared["connection"]

    async def exec_async(self, conn):
        return await conn.query("SELECT * FROM data")

    async def post_async(self, shared, prep_res, exec_res):
        # 确保关闭连接
        try:
            shared["result"] = exec_res
        finally:
            await shared["connection"].close()
```

## 性能对比表

| 类 | 执行方式 | 适用场景 | 性能 |
|---|---|---|---|
| `AsyncNode` | 单个异步 | 单个异步操作 | - |
| `AsyncBatchNode` | 顺序异步 | 需要顺序保证 | 1x |
| `AsyncParallelBatchNode` | 并行异步 | I/O 密集批处理 | Nx |
| `AsyncFlow` | 顺序流程 | 标准异步工作流 | - |
| `AsyncBatchFlow` | 顺序批量流程 | 需要顺序的批量流程 | 1x |
| `AsyncParallelBatchFlow` | 并行批量流程 | 独立的批量流程 | Nx |
