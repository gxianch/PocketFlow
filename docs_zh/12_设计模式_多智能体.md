# 设计模式：多智能体

## 概述

**多智能体（Multi-Agent）** 是一种由多个专业化 Agent 协同工作的系统模式。每个 Agent 负责特定领域的任务，通过通信和协调完成复杂目标。

## 多智能体架构类型

### 1. 协作式（Collaborative）
多个 Agent 平等协作，共同完成任务。

### 2. 层级式（Hierarchical）
有主 Agent（监督者）协调多个子 Agent。

### 3. 竞争式（Competitive）
多个 Agent 独立产生结果，选择最优解。

## 协作式多智能体

### 基本结构

```
┌─────────────────────────────────────────────────────────────┐
│                     共享存储 (Shared Store)                   │
└─────────────────────────────────────────────────────────────┘
        ▲                    ▲                    ▲
        │                    │                    │
        ▼                    ▼                    ▼
   ┌─────────┐          ┌─────────┐          ┌─────────┐
   │ Agent A │◀────────▶│ Agent B │◀────────▶│ Agent C │
   │ (研究员) │          │ (写作者) │          │ (审核者) │
   └─────────┘          └─────────┘          └─────────┘
```

### 实现示例

```python
from pocketflow import Node, Flow

class ResearcherAgent(Node):
    """研究员 Agent：负责信息收集"""
    def prep(self, shared):
        return shared["task"]

    def exec(self, task):
        prompt = f"""
你是一个研究员。为以下任务收集相关信息：
{task}

返回研究发现的要点列表。
"""
        return call_llm(prompt)

    def post(self, shared, prep_res, exec_res):
        shared["research"] = exec_res
        # 将消息传递给下一个 Agent
        shared["messages"].append({
            "from": "researcher",
            "content": exec_res
        })

class WriterAgent(Node):
    """写作者 Agent：负责内容创作"""
    def prep(self, shared):
        return {
            "task": shared["task"],
            "research": shared["research"]
        }

    def exec(self, data):
        prompt = f"""
你是一个写作者。根据研究资料撰写内容：

任务: {data["task"]}
研究资料: {data["research"]}

请撰写完整的内容。
"""
        return call_llm(prompt)

    def post(self, shared, prep_res, exec_res):
        shared["draft"] = exec_res
        shared["messages"].append({
            "from": "writer",
            "content": exec_res
        })

class ReviewerAgent(Node):
    """审核者 Agent：负责质量审核"""
    def prep(self, shared):
        return shared["draft"]

    def exec(self, draft):
        prompt = f"""
你是一个审核者。审核以下内容的质量：

{draft}

如果需要修改，返回 {{"approved": false, "feedback": "修改建议"}}
如果质量合格，返回 {{"approved": true, "final": "最终版本"}}
"""
        return parse_json(call_llm(prompt))

    def post(self, shared, prep_res, exec_res):
        shared["messages"].append({
            "from": "reviewer",
            "content": exec_res
        })

        if exec_res["approved"]:
            shared["final_output"] = exec_res["final"]
            return "complete"
        else:
            shared["feedback"] = exec_res["feedback"]
            return "revise"

# 构建协作流程
researcher = ResearcherAgent()
writer = WriterAgent()
reviewer = ReviewerAgent()

researcher >> writer >> reviewer
reviewer - "revise" >> writer  # 修改循环
reviewer - "complete" >> complete_node

collaboration_flow = Flow(start=researcher)

# 运行
shared = {
    "task": "写一篇关于人工智能发展的文章",
    "messages": []
}
collaboration_flow.run(shared)
```

## 层级式多智能体（监督者模式）

### 结构

```
                    ┌─────────────┐
                    │  Supervisor │
                    │   (监督者)   │
                    └─────────────┘
                          │
           ┌──────────────┼──────────────┐
           ▼              ▼              ▼
      ┌─────────┐    ┌─────────┐    ┌─────────┐
      │ Worker1 │    │ Worker2 │    │ Worker3 │
      │ (搜索)   │    │ (分析)   │    │ (总结)   │
      └─────────┘    └─────────┘    └─────────┘
```

### 实现示例

```python
class SupervisorAgent(Node):
    """监督者 Agent：分配和协调任务"""
    def prep(self, shared):
        return {
            "task": shared["task"],
            "worker_results": shared.get("worker_results", {}),
            "history": shared.get("supervisor_history", [])
        }

    def exec(self, data):
        prompt = f"""
你是一个项目监督者。管理以下任务的执行：

任务: {data["task"]}
已完成的工作: {data["worker_results"]}
历史决策: {data["history"]}

可用的工作者:
- search_worker: 搜索信息
- analyze_worker: 分析数据
- summarize_worker: 总结内容

决定下一步行动，返回 JSON:
{{"action": "delegate", "worker": "工作者名", "sub_task": "子任务"}}
或
{{"action": "complete", "final_answer": "最终答案"}}
"""
        return parse_json(call_llm(prompt))

    def post(self, shared, prep_res, exec_res):
        if "supervisor_history" not in shared:
            shared["supervisor_history"] = []
        shared["supervisor_history"].append(exec_res)

        if exec_res["action"] == "complete":
            shared["final_answer"] = exec_res["final_answer"]
            return "done"

        shared["current_worker"] = exec_res["worker"]
        shared["current_sub_task"] = exec_res["sub_task"]
        return exec_res["worker"]

class SearchWorker(Node):
    """搜索工作者"""
    def prep(self, shared):
        return shared["current_sub_task"]

    def exec(self, task):
        return web_search(task)

    def post(self, shared, prep_res, exec_res):
        if "worker_results" not in shared:
            shared["worker_results"] = {}
        shared["worker_results"]["search"] = exec_res
        return "report"  # 回报给监督者

class AnalyzeWorker(Node):
    """分析工作者"""
    def prep(self, shared):
        return {
            "task": shared["current_sub_task"],
            "data": shared.get("worker_results", {})
        }

    def exec(self, data):
        prompt = f"分析以下数据：\n{data}"
        return call_llm(prompt)

    def post(self, shared, prep_res, exec_res):
        if "worker_results" not in shared:
            shared["worker_results"] = {}
        shared["worker_results"]["analysis"] = exec_res
        return "report"

class SummarizeWorker(Node):
    """总结工作者"""
    def prep(self, shared):
        return shared.get("worker_results", {})

    def exec(self, data):
        prompt = f"总结以下内容：\n{data}"
        return call_llm(prompt)

    def post(self, shared, prep_res, exec_res):
        if "worker_results" not in shared:
            shared["worker_results"] = {}
        shared["worker_results"]["summary"] = exec_res
        return "report"

# 构建监督者流程
supervisor = SupervisorAgent()
search_worker = SearchWorker()
analyze_worker = AnalyzeWorker()
summarize_worker = SummarizeWorker()

supervisor - "search_worker" >> search_worker - "report" >> supervisor
supervisor - "analyze_worker" >> analyze_worker - "report" >> supervisor
supervisor - "summarize_worker" >> summarize_worker - "report" >> supervisor
supervisor - "done" >> complete_node

supervisor_flow = Flow(start=supervisor)
```

## 竞争式多智能体（投票模式）

### 结构

```
                    ┌─────────────┐
                    │   问题输入   │
                    └─────────────┘
                          │
         ┌────────────────┼────────────────┐
         ▼                ▼                ▼
    ┌─────────┐      ┌─────────┐      ┌─────────┐
    │ Solver1 │      │ Solver2 │      │ Solver3 │
    └─────────┘      └─────────┘      └─────────┘
         │                │                │
         └────────────────┼────────────────┘
                          ▼
                    ┌─────────────┐
                    │    投票器    │
                    └─────────────┘
                          │
                          ▼
                    ┌─────────────┐
                    │   最终答案   │
                    └─────────────┘
```

### 实现示例

```python
from pocketflow import BatchNode

class MultiSolverNode(BatchNode):
    """多个求解器并行生成答案"""
    def prep(self, shared):
        # 返回多个不同的求解策略
        return [
            {"strategy": "analytical", "prompt_style": "严谨分析"},
            {"strategy": "creative", "prompt_style": "创造性思考"},
            {"strategy": "practical", "prompt_style": "实用主义"}
        ]

    def exec(self, config):
        prompt = f"""
使用{config["prompt_style"]}方法解决以下问题：
{self.params["problem"]}

请给出你的答案和推理过程。
"""
        return {
            "strategy": config["strategy"],
            "answer": call_llm(prompt)
        }

    def post(self, shared, prep_res, exec_res):
        shared["candidate_answers"] = exec_res

class VotingNode(Node):
    """投票节点：选择最佳答案"""
    def prep(self, shared):
        return {
            "problem": shared["problem"],
            "candidates": shared["candidate_answers"]
        }

    def exec(self, data):
        candidates_text = "\n\n".join([
            f"[{c['strategy']}] {c['answer']}"
            for c in data["candidates"]
        ])

        prompt = f"""
问题: {data["problem"]}

以下是不同策略生成的答案：
{candidates_text}

请评估这些答案，选择最佳的一个或综合它们的优点。
返回最终答案。
"""
        return call_llm(prompt)

    def post(self, shared, prep_res, exec_res):
        shared["final_answer"] = exec_res

# 构建投票流程
solvers = MultiSolverNode()
voter = VotingNode()

solvers >> voter

voting_flow = Flow(start=solvers)
solvers.set_params({"problem": "如何提高代码质量？"})

shared = {"problem": "如何提高代码质量？"}
voting_flow.run(shared)
```

## 异步多智能体

使用异步节点实现真正的并行 Agent：

```python
from pocketflow import AsyncParallelBatchNode, AsyncFlow

class ParallelAgentsNode(AsyncParallelBatchNode):
    """并行执行多个 Agent"""
    async def prep_async(self, shared):
        return [
            {"agent": "research", "task": "搜索相关资料"},
            {"agent": "code", "task": "编写代码示例"},
            {"agent": "docs", "task": "编写文档"}
        ]

    async def exec_async(self, config):
        prompt = f"""
你是一个{config["agent"]}专家。
任务: {config["task"]}

请完成任务并返回结果。
"""
        return {
            "agent": config["agent"],
            "result": await async_llm_call(prompt)
        }

    async def post_async(self, shared, prep_res, exec_res):
        shared["agent_results"] = {
            r["agent"]: r["result"] for r in exec_res
        }

class AggregatorNode(AsyncNode):
    """聚合多个 Agent 的结果"""
    async def prep_async(self, shared):
        return shared["agent_results"]

    async def exec_async(self, results):
        prompt = f"""
整合以下多个专家的工作成果：

研究结果: {results.get("research", "")}
代码示例: {results.get("code", "")}
文档内容: {results.get("docs", "")}

生成一个统一的输出。
"""
        return await async_llm_call(prompt)

    async def post_async(self, shared, prep_res, exec_res):
        shared["final_output"] = exec_res

# 并行 Agent 流程
parallel_agents = ParallelAgentsNode()
aggregator = AggregatorNode()

parallel_agents >> aggregator

async_flow = AsyncFlow(start=parallel_agents)

# 运行
async def main():
    shared = {"task": "创建一个 Python Web API"}
    await async_flow.run_async(shared)
    print(shared["final_output"])

asyncio.run(main())
```

## Agent 间通信

### 消息队列模式

```python
class MessageQueue:
    """简单的消息队列"""
    def __init__(self):
        self.queues = {}

    def send(self, to_agent, message):
        if to_agent not in self.queues:
            self.queues[to_agent] = []
        self.queues[to_agent].append(message)

    def receive(self, agent_name):
        return self.queues.get(agent_name, [])

    def clear(self, agent_name):
        self.queues[agent_name] = []

class CommunicatingAgent(Node):
    """支持通信的 Agent"""
    def prep(self, shared):
        mq = shared["message_queue"]
        messages = mq.receive(self.params["agent_name"])
        return {
            "inbox": messages,
            "context": shared.get("context", {})
        }

    def exec(self, data):
        # 处理收到的消息
        for msg in data["inbox"]:
            self.process_message(msg)

        # 生成响应
        return self.generate_response(data)

    def post(self, shared, prep_res, exec_res):
        mq = shared["message_queue"]

        # 发送消息给其他 Agent
        if exec_res.get("send_to"):
            mq.send(exec_res["send_to"], exec_res["message"])

        return exec_res.get("next_action", "continue")
```

## Taboo 游戏示例

经典的多智能体示例 —— 两个 Agent 玩 Taboo 游戏：

```python
class GuesserAgent(Node):
    """猜词者 Agent"""
    def prep(self, shared):
        return shared.get("clues", [])

    def exec(self, clues):
        if not clues:
            return {"guess": None, "message": "请给我一些提示"}

        prompt = f"""
你在玩 Taboo 游戏。根据以下提示猜测一个词：
提示: {clues}

你的猜测是什么？
"""
        return {"guess": call_llm(prompt)}

    def post(self, shared, prep_res, exec_res):
        shared["guess"] = exec_res["guess"]

        if shared["guess"].lower() == shared["target_word"].lower():
            return "correct"
        return "wrong"

class ClueGiverAgent(Node):
    """提示者 Agent"""
    def prep(self, shared):
        return {
            "target": shared["target_word"],
            "taboo_words": shared["taboo_words"],
            "previous_clues": shared.get("clues", []),
            "wrong_guesses": shared.get("wrong_guesses", [])
        }

    def exec(self, data):
        prompt = f"""
你在玩 Taboo 游戏。
目标词: {data["target"]}
禁止使用的词: {data["taboo_words"]}
之前给过的提示: {data["previous_clues"]}
错误的猜测: {data["wrong_guesses"]}

给出一个新的提示，帮助对方猜出目标词。
不能使用禁止词！
"""
        return call_llm(prompt)

    def post(self, shared, prep_res, exec_res):
        if "clues" not in shared:
            shared["clues"] = []
        shared["clues"].append(exec_res)
        return "guess"

# 游戏流程
guesser = GuesserAgent()
clue_giver = ClueGiverAgent()

clue_giver - "guess" >> guesser
guesser - "wrong" >> clue_giver
guesser - "correct" >> win_node

game = Flow(start=clue_giver)

shared = {
    "target_word": "苹果",
    "taboo_words": ["水果", "红色", "圆形"]
}
game.run(shared)
```

```python

class ResearchAgent: pass   # 只负责研究

# ❌ 避免职责模糊
class GeneralAgent: pass    # 什么都做
```

### 2. 设置迭代限制

```python
class SafeMultiAgent(Node):
    MAX_ROUNDS = 10

    def post(self, shared, prep_res, exec_res):
        rounds = shared.get("rounds", 0) + 1
        if rounds >= self.MAX_ROUNDS:
            return "force_complete"
        shared["rounds"] = rounds
        return "continue"
```
