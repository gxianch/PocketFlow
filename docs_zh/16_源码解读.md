# 源码解读

## 概述

PocketFlow 的核心代码仅 100 行，本文档逐行解析每一行代码的作用。

## 完整源码

```python
import asyncio, warnings, copy, time

class BaseNode:
    def __init__(self): self.params,self.successors={},{}
    def set_params(self,params): self.params=params
    def next(self,node,action="default"):
        if action in self.successors: warnings.warn(f"Overwriting successor for action '{action}'")
        self.successors[action]=node; return node
    def prep(self,shared): pass
    def exec(self,prep_res): pass
    def post(self,shared,prep_res,exec_res): pass
    def _exec(self,prep_res): return self.exec(prep_res)
    def _run(self,shared): p=self.prep(shared); e=self._exec(p); return self.post(shared,p,e)
    def run(self,shared):
        if self.successors: warnings.warn("Node won't run successors. Use Flow.")
        return self._run(shared)
    def __rshift__(self,other): return self.next(other)
    def __sub__(self,action):
        if isinstance(action,str): return _ConditionalTransition(self,action)
        raise TypeError("Action must be a string")

class _ConditionalTransition:
    def __init__(self,src,action): self.src,self.action=src,action
    def __rshift__(self,tgt): return self.src.next(tgt,self.action)

class Node(BaseNode):
    def __init__(self,max_retries=1,wait=0): super().__init__(); self.max_retries,self.wait=max_retries,wait
    def exec_fallback(self,prep_res,exc): raise exc
    def _exec(self,prep_res):
        for self.cur_retry in range(self.max_retries):
            try: return self.exec(prep_res)
            except Exception as e:
                if self.cur_retry==self.max_retries-1: return self.exec_fallback(prep_res,e)
                if self.wait>0: time.sleep(self.wait)

class BatchNode(Node):
    def _exec(self,items): return [super(BatchNode,self)._exec(i) for i in (items or [])]

class Flow(BaseNode):
    def __init__(self,start=None): super().__init__(); self.start_node=start
    def start(self,start): self.start_node=start; return start
    def get_next_node(self,curr,action):
        nxt=curr.successors.get(action or "default")
        if not nxt and curr.successors: warnings.warn(f"Flow ends: '{action}' not found in {list(curr.successors)}")
        return nxt
    def _orch(self,shared,params=None):
        curr,p,last_action =copy.copy(self.start_node),(params or {**self.params}),None
        while curr: curr.set_params(p); last_action=curr._run(shared); curr=copy.copy(self.get_next_node(curr,last_action))
        return last_action
    def _run(self,shared): p=self.prep(shared); o=self._orch(shared); return self.post(shared,p,o)
    def post(self,shared,prep_res,exec_res): return exec_res

class BatchFlow(Flow):
    def _run(self,shared):
        pr=self.prep(shared) or []
        for bp in pr: self._orch(shared,{**self.params,**bp})
        return self.post(shared,pr,None)

class AsyncNode(Node):
    async def prep_async(self,shared): pass
    async def exec_async(self,prep_res): pass
    async def exec_fallback_async(self,prep_res,exc): raise exc
    async def post_async(self,shared,prep_res,exec_res): pass
    async def _exec(self,prep_res):
        for self.cur_retry in range(self.max_retries):
            try: return await self.exec_async(prep_res)
            except Exception as e:
                if self.cur_retry==self.max_retries-1: return await self.exec_fallback_async(prep_res,e)
                if self.wait>0: await asyncio.sleep(self.wait)
    async def run_async(self,shared):
        if self.successors: warnings.warn("Node won't run successors. Use AsyncFlow.")
        return await self._run_async(shared)
    async def _run_async(self,shared): p=await self.prep_async(shared); e=await self._exec(p); return await self.post_async(shared,p,e)
    def _run(self,shared): raise RuntimeError("Use run_async.")

class AsyncBatchNode(AsyncNode,BatchNode):
    async def _exec(self,items): return [await super(AsyncBatchNode,self)._exec(i) for i in items]

class AsyncParallelBatchNode(AsyncNode,BatchNode):
    async def _exec(self,items): return await asyncio.gather(*(super(AsyncParallelBatchNode,self)._exec(i) for i in items))

class AsyncFlow(Flow,AsyncNode):
    async def _orch_async(self,shared,params=None):
        curr,p,last_action =copy.copy(self.start_node),(params or {**self.params}),None
        while curr: curr.set_params(p); last_action=await curr._run_async(shared) if isinstance(curr,AsyncNode) else curr._run(shared); curr=copy.copy(self.get_next_node(curr,last_action))
        return last_action
    async def _run_async(self,shared): p=await self.prep_async(shared); o=await self._orch_async(shared); return await self.post_async(shared,p,o)
    async def post_async(self,shared,prep_res,exec_res): return exec_res

class AsyncBatchFlow(AsyncFlow,BatchFlow):
    async def _run_async(self,shared):
        pr=await self.prep_async(shared) or []
        for bp in pr: await self._orch_async(shared,{**self.params,**bp})
        return await self.post_async(shared,pr,None)

class AsyncParallelBatchFlow(AsyncFlow,BatchFlow):
    async def _run_async(self,shared):
        pr=await self.prep_async(shared) or []
        await asyncio.gather(*(self._orch_async(shared,{**self.params,**bp}) for bp in pr))
        return await self.post_async(shared,pr,None)
```

## 逐行解析

### 第 1 行：导入

```python
import asyncio, warnings, copy, time
```

仅使用 4 个 Python 标准库模块：
- `asyncio`: 异步编程支持
- `warnings`: 警告提示
- `copy`: 浅拷贝对象
- `time`: 同步等待（重试间隔）

**设计理念**：零外部依赖，仅使用标准库。

### 第 3-17 行：BaseNode 类

```python
class BaseNode:
    def __init__(self): self.params,self.successors={},{}
```

**构造函数**：初始化两个空字典
- `params`: 存储节点参数
- `successors`: 存储后继节点映射

```python
    def set_params(self,params): self.params=params
```

**设置参数**：简单赋值，用于 Flow 传递参数给节点。

```python
    def next(self,node,action="default"):
        if action in self.successors: warnings.warn(f"Overwriting successor for action '{action}'")
        self.successors[action]=node; return node
```

**添加后继**：
- 如果 action 已存在，发出警告
- 将节点添加到 successors 字典
- 返回节点（支持链式调用）

```python
    def prep(self,shared): pass
    def exec(self,prep_res): pass
    def post(self,shared,prep_res,exec_res): pass
```

**三阶段方法**：默认为空实现，由子类覆写。

```python
    def _exec(self,prep_res): return self.exec(prep_res)
```

**内部执行**：简单调用 `exec`，可被子类覆写以添加重试等逻辑。

```python
    def _run(self,shared): p=self.prep(shared); e=self._exec(p); return self.post(shared,p,e)
```

**内部运行**：执行三阶段流程
1. `prep` 准备数据
2. `_exec` 执行逻辑
3. `post` 后处理并返回 action

```python
    def run(self,shared):
        if self.successors: warnings.warn("Node won't run successors. Use Flow.")
        return self._run(shared)
```

**公开运行**：
- 如果有后继节点，警告应使用 Flow
- 调用内部 `_run`

```python
    def __rshift__(self,other): return self.next(other)
```

**>> 运算符**：`node_a >> node_b` 等同于 `node_a.next(node_b)`

```python
    def __sub__(self,action):
        if isinstance(action,str): return _ConditionalTransition(self,action)
        raise TypeError("Action must be a string")
```

**- 运算符**：返回辅助对象，支持 `node - "action" >> target` 语法

### 第 19-21 行：_ConditionalTransition 辅助类

```python
class _ConditionalTransition:
    def __init__(self,src,action): self.src,self.action=src,action
    def __rshift__(self,tgt): return self.src.next(tgt,self.action)
```

**作用**：实现条件转换 DSL
- 存储源节点和 action
- `>>` 时调用源节点的 `next` 方法

**使用**：`node_a - "search" >> node_b`
1. `node_a - "search"` 返回 `_ConditionalTransition(node_a, "search")`
2. `>> node_b` 调用 `node_a.next(node_b, "search")`

### 第 23-30 行：Node 类

```python
class Node(BaseNode):
    def __init__(self,max_retries=1,wait=0): super().__init__(); self.max_retries,self.wait=max_retries,wait
```

**构造函数**：添加重试参数

```python
    def exec_fallback(self,prep_res,exc): raise exc
```

**降级处理**：默认重新抛出异常

```python
    def _exec(self,prep_res):
        for self.cur_retry in range(self.max_retries):
            try: return self.exec(prep_res)
            except Exception as e:
                if self.cur_retry==self.max_retries-1: return self.exec_fallback(prep_res,e)
                if self.wait>0: time.sleep(self.wait)
```

**重试逻辑**：
1. 循环尝试 max_retries 次
2. 成功则返回结果
3. 失败时：
   - 最后一次调用 `exec_fallback`
   - 否则等待后继续

### 第 35-47 行：Flow 类

```python
class Flow(BaseNode):
    def __init__(self,start=None): super().__init__(); self.start_node=start
    def start(self,start): self.start_node=start; return start
```

**构造和设置起始节点**

```python
    def get_next_node(self,curr,action):
        nxt=curr.successors.get(action or "default")
        if not nxt and curr.successors: warnings.warn(f"Flow ends: '{action}' not found in {list(curr.successors)}")
        return nxt
```

**获取下一节点**：
- 从 successors 中查找（默认 "default"）
- 找不到时发出警告

```python
    def _orch(self,shared,params=None):
        curr,p,last_action =copy.copy(self.start_node),(params or {**self.params}),None
        while curr: curr.set_params(p); last_action=curr._run(shared); curr=copy.copy(self.get_next_node(curr,last_action))
        return last_action
```

**编排执行**（核心）：
1. 复制起始节点（避免状态污染）
2. 循环直到没有下一个节点：
   - 设置参数
   - 运行节点
   - 获取下一个节点（也复制）
3. 返回最后的 action

```python
    def _run(self,shared): p=self.prep(shared); o=self._orch(shared); return self.post(shared,p,o)
    def post(self,shared,prep_res,exec_res): return exec_res
```

**运行**：执行 prep → 编排 → post

## 设计亮点

### 1. 极简主义
100 行实现完整功能，无冗余代码。

### 2. 关注点分离
prep/exec/post 三阶段模型，职责清晰。

### 3. 运算符重载
`>>` 和 `-` 提供直观的 DSL 语法。

### 4. 节点复制
`copy.copy` 避免状态污染。

### 5. 多继承复用
通过多继承组合功能（如 `AsyncBatchNode(AsyncNode, BatchNode)`）。

### 6. 渐进式异步
同步和异步节点可在同一 Flow 中混用。
