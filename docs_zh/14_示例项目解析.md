## 示例1：聊天机器人（pocketflow-chat）

### 功能
一个简单的命令行聊天机器人，维护对话历史，支持多轮对话。

### 代码解析

```python
from pocketflow import Node, Flow
from utils import call_llm

class ChatNode(Node):
    def prep(self, shared):
        # 首次运行时初始化消息历史
        if "messages" not in shared:
            shared["messages"] = []
            print("Welcome to the chat! Type 'exit' to end.")

        # 获取用户输入
        user_input = input("\nYou: ")

        # 检查是否退出
        if user_input.lower() == 'exit':
            return None

        # 添加用户消息到历史
        shared["messages"].append({"role": "user", "content": user_input})

        # 返回所有消息供 LLM 使用
        return shared["messages"]

    def exec(self, messages):
        if messages is None:
            return None

        # 调用 LLM，传入完整对话历史
        response = call_llm(messages)
        return response

    def post(self, shared, prep_res, exec_res):
        if prep_res is None or exec_res is None:
            print("\nGoodbye!")
            return None  # 结束对话

        # 打印助手回复
        print(f"\nAssistant: {exec_res}")

        # 添加助手消息到历史
        shared["messages"].append({"role": "assistant", "content": exec_res})

        # 返回 "continue" 继续循环
        return "continue"

# 创建自循环流程
chat_node = ChatNode()
chat_node - "continue" >> chat_node  # 自循环

flow = Flow(start=chat_node)

# 运行
if __name__ == "__main__":
    shared = {}
    flow.run(shared)
```

### 关键技巧

1. **自循环实现多轮对话**：
   ```python
   chat_node - "continue" >> chat_node
   ```

2. **对话历史管理**：
   - 在 `prep` 中初始化和读取
   - 在 `post` 中追加新消息

3. **优雅退出**：
   - 返回 `None` 终止循环

### 流程图

```
┌───────────────┐
│   ChatNode    │◀──────┐
│               │       │
│ prep: 获取输入 │       │ "continue"
│ exec: 调用 LLM │       │
│ post: 存储回复 │───────┘
│               │
│   (exit)      │──────▶ 结束
└───────────────┘
```

## 示例2：研究型 Agent（pocketflow-agent）

### 功能
一个能够自主搜索网络并回答问题的 Agent。

### 核心节点

#### DecideAction 节点

```python
class DecideAction(Node):
    def prep(self, shared):
        context = shared.get("context", "No previous search")
        question = shared["question"]
        return question, context

    def exec(self, inputs):
        question, context = inputs

        prompt = f"""
### CONTEXT
You are a research assistant that can search the web.
Question: {question}
Previous Research: {context}

### ACTION SPACE
[1] search - Look up more information
[2] answer - Answer with current knowledge

Decide the next action. Return YAML format.
"""
        response = call_llm(prompt)
        # 解析 YAML 响应
        decision = yaml.safe_load(response)
        return decision

    def post(self, shared, prep_res, exec_res):
        if exec_res["action"] == "search":
            shared["search_query"] = exec_res["search_query"]
            return "search"
        else:
            shared["context"] = exec_res["answer"]
            return "answer"
```

#### SearchWeb 节点

```python
class SearchWeb(Node):
    def prep(self, shared):
        return shared["search_query"]

    def exec(self, search_query):
        results = search_web_duckduckgo(search_query)
        return results

    def post(self, shared, prep_res, exec_res):
        # 累积搜索结果到上下文
        previous = shared.get("context", "")
        shared["context"] = previous + "\n\nSEARCH: " + \
            shared["search_query"] + "\nRESULTS: " + exec_res
        return "decide"  # 回到决策节点
```

#### AnswerQuestion 节点

```python
class AnswerQuestion(Node):
    def prep(self, shared):
        return shared["question"], shared.get("context", "")

    def exec(self, inputs):
        question, context = inputs
        prompt = f"""
Based on the research, answer the question.
Question: {question}
Research: {context}
"""
        return call_llm(prompt)

    def post(self, shared, prep_res, exec_res):
        shared["answer"] = exec_res
        return "done"
```

### 流程构建

```python
# 创建节点
decide = DecideAction()
search = SearchWeb()
answer = AnswerQuestion()

# 定义流程
decide - "search" >> search - "decide" >> decide
decide - "answer" >> answer

flow = Flow(start=decide)
```

### 流程图

```
              ┌─────────────────────────┐
              │                         │
              ▼                         │
        ┌───────────┐                   │
        │  Decide   │───"search"───┐    │
        │  Action   │              │    │
        └───────────┘              ▼    │
              │            ┌───────────┐│
              │            │  Search   ││
           "answer"        │   Web     ││
              │            └───────────┘│
              ▼                  │      │
        ┌───────────┐            │      │
        │  Answer   │      "decide"     │
        │ Question  │            └──────┘
        └───────────┘
```

## 示例3：RAG 检索增强（pocketflow-rag）

### 功能
实现完整的 RAG 系统，包括离线索引和在线查询。

### 离线索引流程

#### 文档分块节点

```python
class ChunkDocumentsNode(BatchNode):
    def prep(self, shared):
        return shared["texts"]

    def exec(self, text):
        # 对单个文本进行分块
        return fixed_size_chunk(text)

    def post(self, shared, prep_res, exec_res_list):
        # 展平所有分块
        all_chunks = []
        for chunks in exec_res_list:
            all_chunks.extend(chunks)
        shared["texts"] = all_chunks
        print(f"✅ Created {len(all_chunks)} chunks")
        return "default"
```

#### 嵌入节点

```python
class EmbedDocumentsNode(BatchNode):
    def prep(self, shared):
        return shared["texts"]

    def exec(self, text):
        return get_embedding(text)

    def post(self, shared, prep_res, exec_res_list):
        embeddings = np.array(exec_res_list, dtype=np.float32)
        shared["embeddings"] = embeddings
        return "default"
```

#### 索引创建节点

```python
class CreateIndexNode(Node):
    def prep(self, shared):
        return shared["embeddings"]

    def exec(self, embeddings):
        dimension = embeddings.shape[1]
        index = faiss.IndexFlatL2(dimension)
        index.add(embeddings)
        return index

    def post(self, shared, prep_res, exec_res):
        shared["index"] = exec_res
        return "default"
```

### 在线查询流程

#### 查询嵌入节点

```python
class EmbedQueryNode(Node):
    def prep(self, shared):
        return shared["query"]

    def exec(self, query):
        query_embedding = get_embedding(query)
        return np.array([query_embedding], dtype=np.float32)

    def post(self, shared, prep_res, exec_res):
        shared["query_embedding"] = exec_res
        return "default"
```

#### 检索节点

```python
class RetrieveDocumentNode(Node):
    def prep(self, shared):
        return (shared["query_embedding"],
                shared["index"],
                shared["texts"])

    def exec(self, inputs):
        query_embedding, index, texts = inputs
        distances, indices = index.search(query_embedding, k=1)
        best_idx = indices[0][0]
        return {
            "text": texts[best_idx],
            "index": best_idx,
            "distance": distances[0][0]
        }

    def post(self, shared, prep_res, exec_res):
        shared["retrieved_document"] = exec_res
        return "default"
```

#### 生成回答节点

```python
class GenerateAnswerNode(Node):
    def prep(self, shared):
        return shared["query"], shared["retrieved_document"]

    def exec(self, inputs):
        query, retrieved_doc = inputs
        prompt = f"""
Answer the question based on the context:
Question: {query}
Context: {retrieved_doc['text']}
"""
        return call_llm(prompt)

    def post(self, shared, prep_res, exec_res):
        shared["generated_answer"] = exec_res
        return "default"
```

### 流程构建

```python
# 离线索引流程
chunk = ChunkDocumentsNode()
embed = EmbedDocumentsNode()
create_index = CreateIndexNode()

chunk >> embed >> create_index
offline_flow = Flow(start=chunk)

# 在线查询流程
embed_query = EmbedQueryNode()
retrieve = RetrieveDocumentNode()
generate = GenerateAnswerNode()

embed_query >> retrieve >> generate
online_flow = Flow(start=embed_query)
```

### 架构图

```
离线索引:
┌───────────┐     ┌───────────┐     ┌───────────┐
│   Chunk   │────▶│   Embed   │────▶│  Create   │
│ Documents │     │ Documents │     │   Index   │
└───────────┘     └───────────┘     └───────────┘

在线查询:
┌───────────┐     ┌───────────┐     ┌───────────┐
│   Embed   │────▶│  Retrieve │────▶│ Generate  │
│   Query   │     │  Document │     │  Answer   │
└───────────┘     └───────────┘     └───────────┘
```
